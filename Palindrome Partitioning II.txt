class Solution:
    def minCut(self, s: str) -> int:
        n = len(s)

        palindrome = [[False] * n for _ in range(n)]
        cuts = [0] * n

        for i in range(n):
            min_cut = i

            for j in range(i + 1):
                if s[j] == s[i] and (i - j <= 1 or palindrome[j + 1][i - 1]):
                    palindrome[j][i] = True

                    if j == 0:
                        min_cut = 0
                    else:
                        min_cut = min(min_cut, cuts[j - 1] + 1)

            cuts[i] = min_cut

        return cuts[-1]
